class MeuBLECallback: public BLECharacteristicCallbacks {
  void onWrite(BLECharacteristic *pCharacteristic) {
    String rx = pCharacteristic->getValue().c_str();
    rx.trim();
    if (rx == "HOME") resetarPara90();
    else if (rx.startsWith("S")) {
      int sep = rx.indexOf(':');
      if (sep > 0) {
        int idx = rx.substring(1, sep).toInt();
        int ang = constrain(rx.substring(sep + 1).toInt(), 0, 180);
        if (idx >= 0 && idx < qtdServosAtivos) {
          posAtual[idx] = ang;
          bracoServos[idx].write(ang);
        }
      }
    }
  }
};

void setup() {
  Serial.begin(115200);
  pinMode(PINO_SENSE_6, INPUT_PULLUP);
  pinMode(PINO_SENSE_8, INPUT_PULLUP);

  if (digitalRead(PINO_SENSE_8) == LOW) qtdServosAtivos = 8;
  else if (digitalRead(PINO_SENSE_6) == LOW) qtdServosAtivos = 6;
  else qtdServosAtivos = 4;

  memoriaNVS.begin("garra_robot", false);
  totalPassosSalvos = memoriaNVS.getInt("total", 0);

  ESP32PWM::allocateTimer(0);
  ESP32PWM::allocateTimer(1);
  for (int i = 0; i < qtdServosAtivos; i++) {
    bracoServos[i].setPeriodHertz(50);
    bracoServos[i].attach(PINO_SERVOS[i], 500, 2400);
    bracoServos[i].write(90);
    posAtual[i] = 90;
    delay(150);
  }

  WiFi.mode(WIFI_AP);
  WiFi.softAP(AP_SSID, AP_PASS);

  BLEDevice::init("Garra_BLE_Control");
  BLEServer *pServer = BLEDevice::createServer();
  BLEService *pService = pServer->createService(SERVICE_UUID);
  BLECharacteristic *pChar = pService->createCharacteristic(CHARACTERISTIC_UUID, BLECharacteristic::PROPERTY_READ | BLECharacteristic::PROPERTY_WRITE);
  pChar->setCallbacks(new MeuBLECallback());
  pService->start();
  BLEDevice::getAdvertising()->start();

  server.on("/", []() { server.send_P(200, "text/html", INDEX_HTML); });
  server.on("/status", []() {
    String j = "{\"qtd\":" + String(qtdServosAtivos) + ",\"pos\":[";
    for (int i = 0; i < qtdServosAtivos; i++) j += String(posAtual[i]) + (i < qtdServosAtivos - 1 ? "," : "");
    j += "],\"telemetria\":\"" + gerarTelemetria() + "\"}";
    server.send(200, "application/json", j);
  });
  server.on("/set", []() {
    int s = server.arg("s").toInt();
    int a = constrain(server.arg("a").toInt(), 0, 180);
    if (s >= 0 && s < qtdServosAtivos) { posAtual[s] = a; bracoServos[s].write(a); }
    server.send(200, "text/plain", gerarTelemetria());
  });
  server.on("/cmd", []() {
    String c = server.arg("c");
    if (c == "home") resetarPara90();
    else if (c == "save" && totalPassosSalvos < MAX_PASSOS) {
      for (int i = 0; i < 8; i++) memoriaNVS.putInt(("p" + String(totalPassosSalvos) + "s" + String(i)).c_str(), posAtual[i]);
      memoriaNVS.putInt("total", ++totalPassosSalvos);
    } else if (c == "play") executarSeq();
    else if (c == "clear") { totalPassosSalvos = 0; memoriaNVS.putInt("total", 0); }
    server.send(200, "text/plain", gerarTelemetria());
  });
  server.begin();
}

void loop() {
  server.handleClient();
  if (millis() - ultimoTempoSerial >= 500) {
    ultimoTempoSerial = millis();
    Serial.println(gerarTelemetria());
  }
}

void resetarPara90() {
  for (int i = 0; i < qtdServosAtivos; i++) { posAtual[i] = 90; bracoServos[i].write(90); delay(80); }
}

void executarSeq() {
  for (int p = 0; p < totalPassosSalvos; p++) {
    bool mov = true;
    while (mov) {
      mov = false;
      for (int i = 0; i < qtdServosAtivos; i++) {
        int alvo = memoriaNVS.getInt(("p" + String(p) + "s" + String(i)).c_str(), 90);
        if (posAtual[i] != alvo) { posAtual[i] += (alvo > posAtual[i]) ? 1 : -1; bracoServos[i].write(posAtual[i]); mov = true; }
      }
      delay(15);
    }
    delay(350);
  }
}

String gerarTelemetria() {
  float rMax = L2_BRACO + L3_ANTEBRACO;
  float rO = radians(posAtual[1]), rC = radians(posAtual[2] - 90.0);
  float alt = max(0.0f, L1_BASE_ALTURA + L2_BRACO * sin(rO) + L3_ANTEBRACO * sin(rO + rC));
  float domo = abs(L2_BRACO * cos(rO) + L3_ANTEBRACO * cos(rO + rC));
  String out = String(PLACA_NOME) + " | servo:" + String(SERVO_MODELO) + " ang:0-180 carga:" + String(SERVO_FORCA_GRAMAS) + "g | Alt:" + String(alt, 1) + "mm Domo:" + String(domo, 1) + "/" + String(rMax, 0) + "mm | Mem:" + String(totalPassosSalvos) + " | [";
  for (int i = 0; i < qtdServosAtivos; i++) out += "S" + String(i + 1) + ":" + String(posAtual[i]) + (i < qtdServosAtivos - 1 ? " " : "");
  return out + "]";
}

Como Cada Parte do Código Funciona

1. Diretiva #if CONFIG_IDF_TARGET_ESP32C3

Identifica automaticamente no momento da compilação se você selecionou o ESP32-C3 ou o DevKit V1, evitando conflitos com o pino de boot (GPIO 9) do C3.

2. WiFi.softAP() & WebServer

Liga o rádio Wi-Fi em modo Ponto de Acesso (roteador próprio), servindo a interface HTML armazenada na memória Flash (PROGMEM) sem gastar memória RAM.

3. BLEDevice & Callbacks

Ativa simultaneamente o Bluetooth Low Energy. Quando o smartphone envia comandos como S0:45 ou HOME, o evento onWrite movimenta o eixo instantaneamente.

4. Memória NVS (Preferences)

Substitui a antiga EEPROM por pares chave-valor na memória Flash do ESP32, preservando até 15 passos completos mesmo após cortes de energia.
